CREATE DATABASE IF NOT EXISTS Naturista;
USE Naturista;

-- =======================================================================================================================================================
-- TABLAS ESQUEMA  INVENTARIO
-- =======================================================================================================================================================
-- =========================================
-- TABLA: CATEGORIAS
-- Categorias de Producto
-- =========================================
DROP TABLE IF EXISTS CATEGORIAS;
CREATE TABLE CATEGORIAS (
    ID_CATEGORIA		VARCHAR(10)   	NOT NULL PRIMARY KEY COMMENT 'Código único de la categoría',
    NOMBRE_CATEGORIA	VARCHAR(100)  	NOT NULL COMMENT 'Nombre de la categoría',
    DESCRIPCION			VARCHAR(300)  	COMMENT 'Descripción detallada de la categoría',
    ESTADO				CHAR(1)       	DEFAULT 'A' COMMENT 'Estado de la categoría (A=Activo, I=Inactivo, S= Suspendido, P=Pendiente, E= En Espera, O=Obsoleto, R=Reservado, T=Test)',
    FECHA_CREACION		TIMESTAMP		DEFAULT CURRENT_TIMESTAMP COMMENT 'Fecha y hora de creación del registro',
    FECHA_ACTUALIZACION	TIMESTAMP   	DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP COMMENT 'Última fecha de actualización del registro',
    CONSTRAINT chk_estado_cat CHECK (ESTADO IN ('A','I','S','P','E','O','R','T'))
) COMMENT='Tabla de categorías de productos para clasificación en inventario';

-- =========================================
-- TABLA: SUBCATEGORIAS
-- SubCategorias de Producto
-- =========================================
DROP TABLE IF EXISTS SUBCATEGORIAS;
CREATE TABLE SUBCATEGORIAS (
    ID_SUBCATEGORIA    	VARCHAR(10)   	NOT NULL PRIMARY KEY COMMENT 'Código único de la subcategoría',
    ID_CATEGORIA       	VARCHAR(10)   	NOT NULL COMMENT 'Código de la categoría padre',
    NOMBRE             	VARCHAR(100)  	NOT NULL COMMENT 'Nombre de la subcategoría',
    DESCRIPCION        	VARCHAR(300)  	COMMENT 'Descripción detallada de la subcategoría',
    ESTADO             	CHAR(1)       	DEFAULT 'A' COMMENT 'Estado de la subcategoría (A=Activo, I=Inactivo, S= Suspendido, P=Pendiente, E= En Espera, O=Obsoleto, R=Reservado, T=Test)',
    FECHA_CREACION    	TIMESTAMP     	DEFAULT CURRENT_TIMESTAMP COMMENT 'Fecha y hora de creación del registro',
    FECHA_ACTUALIZACION	TIMESTAMP		DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP COMMENT 'Última fecha de actualización del registro',
    CONSTRAINT chk_estado_sct CHECK (ESTADO IN ('A','I','S','P','E','O','R','T')),
    CONSTRAINT fk_categoria FOREIGN KEY (ID_CATEGORIA) REFERENCES CATEGORIAS(ID_CATEGORIA) ON DELETE CASCADE
) COMMENT='Tabla de subcategorías de productos, vinculadas a categorías principales';

-- =========================================
-- TABLA: STOCK
-- Stocks Maximos y Minimos para cada tipo de producto
-- =========================================
DROP TABLE IF EXISTS STOCK;
CREATE TABLE STOCK (
    ID_STOCK           	VARCHAR(10)   	NOT NULL PRIMARY KEY COMMENT 'Código único de stock',
    TIPO_STOCK         	CHAR(1)       	NOT NULL COMMENT 'Tipo de stock (M=Mínimo, X=Máximo)',
    AMBITO             	VARCHAR(50)   	NOT NULL COMMENT 'Ámbito o categoría del stock (General, Aceites, Colágeno, etc.)',
    DESCRIPCION        	VARCHAR(200)  	COMMENT 'Descripción detallada del stock',
    CANTIDAD           	INT           	NOT NULL COMMENT 'Cantidad definida para el stock',
    ESTADO             	CHAR(1)       	DEFAULT 'A' COMMENT 'Estado del registro (A=Activo, I=Inactivo, S= Suspendido, P=Pendiente, E= En Espera, O=Obsoleto, R=Reservado, T=Test)',
    FECHA_CREACION		TIMESTAMP     	DEFAULT CURRENT_TIMESTAMP COMMENT 'Fecha y hora de creación del registro',
    FECHA_ACTUALIZACION	TIMESTAMP		DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP COMMENT 'Última fecha de actualización del registro',
    CONSTRAINT chk_tipo_stock CHECK (TIPO_STOCK IN ('M','X')),
    CONSTRAINT chk_estado_stock CHECK (ESTADO IN ('A','I','S','P','E','O','R','T'))
) COMMENT='Tabla de parámetros de stock mínimo y máximo por ámbito o categoría';

-- =========================================
-- TABLA: PRODUCTOS
-- Productos del Inventario
-- =========================================
DROP TABLE IF EXISTS PRODUCTOS;
CREATE TABLE PRODUCTOS (
    ID_PRODUCTO			VARCHAR(20)		NOT NULL PRIMARY KEY COMMENT 'Código único del producto',
    NOMBRE_PRODUCTO		VARCHAR(200)	NOT NULL COMMENT 'Nombre completo del producto',
    PVP					DECIMAL(10,3)	NOT NULL COMMENT 'Precio de venta al público',
    DESCRIPCION			VARCHAR(500)	COMMENT 'Descripción detallada del producto',
    ID_SUBCATEGORIA		VARCHAR(10)		NOT NULL COMMENT 'Código de subcategoría según catálogo definido',
    ETIQUETAS			VARCHAR(300)	COMMENT 'Palabras clave para búsqueda y clasificación',
    UNIDAD_MEDIDA		VARCHAR(10)		COMMENT 'Unidad de medida (ml, g, cap, etc.)',
    PESO_VOLUMEN		INT				COMMENT 'Valor numérico del peso o volumen',
    ID_LABORATORIO		VARCHAR(20)		COMMENT 'Identificador del laboratorio fabricante',
    CODIGO_BARRAS		VARCHAR(50)		COMMENT 'Código de barras (EAN/UPC), se usa VARCHAR para conservar ceros iniciales',
    FECHA_CADUCIDAD		DATE			COMMENT 'Fecha de caducidad del producto',
    ID_STOCK_MIN		VARCHAR(20)		DEFAULT 0 COMMENT 'Stock mínimo recomendado',
    ID_STOCK_MAX		VARCHAR(20)		DEFAULT 0 COMMENT 'Stock máximo recomendado',
    ID_PROVEEDOR		VARCHAR(20)		COMMENT 'Identificador del proveedor',
    FECHA_CREACION		TIMESTAMP 		DEFAULT CURRENT_TIMESTAMP COMMENT 'Fecha y hora de creación del registro',
    FECHA_ACTUALIZACION	TIMESTAMP 		DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP COMMENT 'Última fecha de actualización del registro',
    ESTADO             	CHAR(1) 		DEFAULT 'A' COMMENT 'Estado del producto (A=Activo, I=Inactivo, S= Suspendido, P=Pendiente, E= En Espera, O=Obsoleto, R=Reservado, T=Test)',
    CONSTRAINT chk_estado CHECK (ESTADO IN ('A','I','S','P','E','O','R','T')),
    CONSTRAINT fk_producto_subcategoria FOREIGN KEY (ID_SUBCATEGORIA) REFERENCES SUBCATEGORIAS(ID_SUBCATEGORIA) ON DELETE RESTRICT
) COMMENT='Tabla de productos con información de inventario, clasificación y auditoría';

CREATE TABLE MARCAS (

    DESCRIPCION			VARCHAR(300) COMMENT 'Descripcion de la Marca',

    FECHA_CREACION 		TIMESTAMP DEFAULT CURRENT_TIMESTAMP COMMENT 'Fecha de Creacion',
    FECHA_ACTUALIZACION TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP COMMENT 'Fecha de Actualizacion',
    CONSTRAINT chk_estado_marca CHECK (ESTADO IN ('A','I','S','P','E','O','R','T'))
);
-- ============================================
-- Tabla: LABORATORIOS
-- Propósito: Registrar los laboratorios fabricantes de productos
-- ============================================
DROP TABLE IF EXISTS LABORATORIOS;
CREATE TABLE LABORATORIOS (
    ID_LABORATORIO      INT AUTO_INCREMENT PRIMARY KEY COMMENT 'Clave Primaria',
    NOMBRE              VARCHAR(250) NOT NULL COMMENT 'Nombre del Laboratorio',
    DESCRIPCION			VARCHAR(300) COMMENT 'Descripcion del Laboratorio',
    ESTADO              CHAR(1) DEFAULT 'A' COMMENT 'Estado del laboratorio (A=Activo, I=Inactivo, S= Suspendido, P=Pendiente, E= En Espera, O=Obsoleto, R=Reservado, T=Test)',
    FECHA_CREACION      TIMESTAMP DEFAULT CURRENT_TIMESTAMP COMMENT 'Fecha de Creación',
    FECHA_ACTUALIZACION TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP COMMENT 'Fecha de Actualización',
    CONSTRAINT chk_estado_lab CHECK (ESTADO IN ('A','I','S','P','E','O','R','T'))
);
-- =========================================
-- TABLA: SUCURSALES
-- Informacion de Sucursales
-- =========================================
DROP TABLE IF EXISTS SUCURSALES;
CREATE TABLE SUCURSALES (
    ID_SUCURSAL        	VARCHAR(10)   NOT NULL PRIMARY KEY COMMENT 'Código único de la sucursal',
    NOMBRE_SUCURSAL    	VARCHAR(150)  NOT NULL COMMENT 'Nombre de la sucursal',
    DIRECCION_SUCURSAL 	VARCHAR(300)  NOT NULL COMMENT 'Dirección física de la sucursal',
    TELEFONO           	VARCHAR(20)   COMMENT 'Teléfono de contacto de la sucursal',
    EMAIL              	VARCHAR(100)  COMMENT 'Correo electrónico de la sucursal',
    RESPONSABLE        	VARCHAR(100)  COMMENT 'Nombre del responsable o administrador de la sucursal',
    RUC_SUCURSAL		VARCHAR(13)	  NOT NULL COMMENT 'Ruc de la Sucursal o Empresa',
    ESTADO             	CHAR(1)       DEFAULT 'A' COMMENT 'Estado de la sucursal (A=Activo, I=Inactivo, S= Suspendido, P=Pendiente, E= En Espera, O=Obsoleto, R=Reservado, T=Test)',
    FECHA_CREACION     	TIMESTAMP     DEFAULT CURRENT_TIMESTAMP COMMENT 'Fecha y hora de creación del registro',
    FECHA_ACTUALIZACION TIMESTAMP    DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP COMMENT 'Última fecha de actualización del registro',
    CONSTRAINT chk_estado_sucursal CHECK (ESTADO IN ('A','I','S','P','E','O','R','T'))
) COMMENT='Tabla de sucursales donde se gestionan productos e inventarios';

-- =========================================
-- TABLA: PRODUCTOS_SUCURSAL
-- Registro de Productos por Sucursal
-- =========================================
DROP TABLE IF EXISTS PRODUCTOS_SUCURSAL;
CREATE TABLE PRODUCTOS_SUCURSAL (
    ID_PRODUCTO        	VARCHAR(20)   	NOT NULL COMMENT 'Código del producto',
    ID_SUCURSAL        	VARCHAR(10)   	NOT NULL COMMENT 'Código de la sucursal',
    CANTIDAD           	INT           	NOT NULL DEFAULT 0 COMMENT 'Cantidad disponible en la sucursal',
    ESTADO             	CHAR(1)       	DEFAULT 'A' COMMENT 'Estado del registro (A=Activo, I=Inactivo, S= Suspendido, P=Pendiente, E= En Espera, O=Obsoleto, R=Reservado, T=Test)',
    FECHA_CREACION     	TIMESTAMP     	DEFAULT CURRENT_TIMESTAMP COMMENT 'Fecha y hora de creación del registro',
    FECHA_ACTUALIZACION	TIMESTAMP		DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP COMMENT 'Última fecha de actualización del registro',
    CONSTRAINT chk_estado_prod_sucursal CHECK (ESTADO IN ('A','I','S','P','E','O','R','T')),
    CONSTRAINT pk_prod_sucursal PRIMARY KEY (ID_PRODUCTO, ID_SUCURSAL),
    CONSTRAINT fk_prod_sucursal_producto FOREIGN KEY (ID_PRODUCTO) REFERENCES PRODUCTOS(ID_PRODUCTO) ON DELETE CASCADE,
    CONSTRAINT fk_prod_sucursal_sucursal FOREIGN KEY (ID_SUCURSAL) REFERENCES SUCURSALES(ID_SUCURSAL) ON DELETE CASCADE
) COMMENT='Tabla que relaciona productos con las sucursales y su inventario disponible';

-- =========================================
-- TABLA: BENEFICIOS_SINTOMAS
-- Catálogo de Beneficios y Sintomas asociados a productos
-- =========================================
DROP TABLE IF EXISTS BENEFICIOS_SINTOMAS;
CREATE TABLE BENEFICIOS_SINTOMAS (
    ID_SINT_BENEF       	VARCHAR(10)   	NOT NULL PRIMARY KEY COMMENT 'Código único del síntoma o beneficio',
    NOMBRE_SINT_BENEF   	VARCHAR(150)  	NOT NULL COMMENT 'Nombre del síntoma o beneficio',
    DESCRIPCION_SINT_BENEF 	VARCHAR(500)	COMMENT 'Descripción detallada del síntoma o beneficio',
    ESTADO              	CHAR(1)       	DEFAULT 'A' COMMENT 'Estado del registro (A=Activo, I=Inactivo, S= Suspendido, P=Pendiente, E= En Espera, O=Obsoleto, R=Reservado, T=Test)',
    FECHA_CREACION      	TIMESTAMP     	DEFAULT CURRENT_TIMESTAMP COMMENT 'Fecha y hora de creación del registro',
    FECHA_ACTUALIZACION 	TIMESTAMP     	DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP COMMENT 'Última fecha de actualización del registro',
    CONSTRAINT chk_estado_ben CHECK (ESTADO IN ('A','I','S','P','E','O','R','T'))
) COMMENT='Tabla de síntomas y beneficios asociados a productos naturales';

-- =========================================
-- TABLA: PRODUCTOS_BENEFICIOS
-- Tabla intermedia que relaciona productos con sus beneficios o síntomas asociados
-- =========================================
DROP TABLE IF EXISTS PRODUCTOS_BENEFICIOS;
CREATE TABLE PRODUCTOS_BENEFICIOS (
    ID_PRODUCTO        	VARCHAR(20)   	NOT NULL COMMENT 'Código del producto',
    ID_SINT_BENEF      	VARCHAR(10)   	NOT NULL COMMENT 'Código del síntoma/beneficio',
    ESTADO             	CHAR(1)       	DEFAULT 'A' COMMENT 'Estado de la relación A=Activo, I=Inactivo, S= Suspendido, P=Pendiente, E= En Espera, O=Obsoleto, R=Reservado, T=Test',
    FECHA_CREACION     	TIMESTAMP     	DEFAULT CURRENT_TIMESTAMP COMMENT 'Fecha y hora de creación del registro',
    FECHA_ACTUALIZACION	TIMESTAMP		DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP COMMENT 'Última fecha de actualización del registro',
    CONSTRAINT chk_estado_prodben CHECK (ESTADO IN ('A','I','S','P','E','O','R','T')),
    CONSTRAINT pk_prod_benef PRIMARY KEY (ID_PRODUCTO, ID_SINT_BENEF),
    CONSTRAINT fk_prod_benef_producto FOREIGN KEY (ID_PRODUCTO) REFERENCES PRODUCTOS(ID_PRODUCTO) ON DELETE CASCADE,
    CONSTRAINT fk_prod_benef_sint FOREIGN KEY (ID_SINT_BENEF) REFERENCES BENEFICIOS_SINTOMAS(ID_SINT_BENEF) ON DELETE RESTRICT
) COMMENT='Tabla intermedia que relaciona productos con sus beneficios o síntomas asociados';

-- =========================================
-- TABLA: PROMOCIONES_COMBOS
-- Tabla unificada de promociones y combos
-- =========================================
DROP TABLE IF EXISTS PROMOCIONES_COMBOS;
CREATE TABLE PROMOCIONES_COMBOS (
    ID_PROMO_COMBO      VARCHAR(10) PRIMARY KEY,
    TIPO                CHAR(1) NOT NULL COMMENT 'P=Promoción, C=Combo',
    DESCRIPCION         VARCHAR(200) NOT NULL,
    CRITERIO            VARCHAR(200) COMMENT 'Condición o regla de aplicación',
    FECHA_INICIO        DATE NOT NULL,
    FECHA_FIN           DATE NOT NULL,
    ESTADO              CHAR(1) DEFAULT 'A' COMMENT 'A=Activo, I=Inactivo, S= Suspendido, P=Pendiente, E= En Espera, O=Obsoleto, R=Reservado, T=Test',
    USOS_MAXIMOS        INT DEFAULT NULL COMMENT 'Número máximo de usos permitidos',
    CANAL               VARCHAR(50) COMMENT 'Canal de aplicación (online, tienda, etc.)',
    FECHA_CREACION      TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    FECHA_ACTUALIZACION TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    CONSTRAINT chk_tipo_promo CHECK (TIPO IN ('P','C')),
    CONSTRAINT chk_estado_promo CHECK (ESTADO IN ('A','I','S','P','E','O','R','T'))
) COMMENT='Tabla unificada de promociones y combos';

-- =========================================
-- TABLA: PROMOCIONES_COMBOS
-- Detalle de productos y descuentos aplicables en promociones
-- =========================================
DROP TABLE IF EXISTS PROMOCIONES_DETALLE;
CREATE TABLE PROMOCIONES_DETALLE (
    ID_DETALLE          	INT AUTO_INCREMENT PRIMARY KEY,
    ID_PROMO_COMBO      	VARCHAR(10) NOT NULL,
    ID_PRODUCTO         	VARCHAR(20) NOT NULL,
    TIPO_DESCUENTO      	CHAR(1) COMMENT 'P=Porcentaje, V=Valor fijo, G=Producto gratis, C=Cantidad (ej. 3x2)',
    VALOR_DESCUENTO     	DECIMAL(10,2) COMMENT 'Porcentaje o valor de descuento',
    CONSTRAINT fk_promocion FOREIGN KEY (ID_PROMO_COMBO) REFERENCES PROMOCIONES_COMBOS(ID_PROMO_COMBO) ON DELETE CASCADE,
    CONSTRAINT fk_producto_prom FOREIGN KEY (ID_PRODUCTO) REFERENCES PRODUCTOS(ID_PRODUCTO) ON DELETE RESTRICT
) COMMENT='Detalle de productos y descuentos aplicables en promociones';

-- =========================================
-- TABLA: COMBOS_DETALLE
-- Detalle de productos incluidos en combos
-- =========================================
DROP TABLE IF EXISTS COMBOS_DETALLE;
CREATE TABLE COMBOS_DETALLE (
    ID_DETALLE          INT AUTO_INCREMENT PRIMARY KEY,
    ID_PROMO_COMBO      VARCHAR(10) NOT NULL,
    ID_PRODUCTO         VARCHAR(20) NOT NULL,
    CANTIDAD_PRODUCTO   INT NOT NULL DEFAULT 1,
    CONSTRAINT fk_combo FOREIGN KEY (ID_PROMO_COMBO) REFERENCES PROMOCIONES_COMBOS(ID_PROMO_COMBO) ON DELETE CASCADE,
    CONSTRAINT fk_producto_combo FOREIGN KEY (ID_PRODUCTO) REFERENCES PRODUCTOS(ID_PRODUCTO) ON DELETE RESTRICT
) COMMENT='Detalle de productos incluidos en combos';

-- =========================================
-- TABLA: AUDITORIA_UPDATE
-- Tabla de auditoría para registros actualizados
-- =========================================
DROP TABLE IF EXISTS AUDITORIA_UPDATE;
CREATE TABLE AUDITORIA_UPDATE (
    ID_LOG          INT AUTO_INCREMENT PRIMARY KEY COMMENT 'Identificador único del log',
    NOMBRE_TABLA    VARCHAR(100) NOT NULL COMMENT 'Nombre de la tabla afectada',
    CLAVE_PRIMARIA  VARCHAR(100) NOT NULL COMMENT 'Identificador del registro modificado',
    CAMPO           VARCHAR(100) NOT NULL COMMENT 'Nombre del campo modificado',
    VALOR_ANTERIOR  VARCHAR(500) COMMENT 'Valor antes de la modificación',
    VALOR_NUEVO     VARCHAR(500) COMMENT 'Valor después de la modificación',
    USUARIO         VARCHAR(100) NOT NULL COMMENT 'Usuario que realizó la modificación',
    FECHA_CAMBIO    TIMESTAMP DEFAULT CURRENT_TIMESTAMP COMMENT 'Fecha y hora del cambio'
) COMMENT='Tabla de auditoría para registros actualizados';

-- =========================================
-- TABLA: AUDITORIA_UPDATE
-- Tabla de auditoría para registros eliminados
-- =========================================
DROP TABLE IF EXISTS AUDITORIA_DELETE;
CREATE TABLE AUDITORIA_DELETE (
	ID_LOG				INT AUTO_INCREMENT PRIMARY KEY COMMENT 'Identificador único del log',
    NOMBRE_TABLA    	VARCHAR(100) NOT NULL COMMENT 'Nombre de la tabla afectada',
    CLAVE_PRIMARIA  	VARCHAR(100) NOT NULL COMMENT 'Identificador del registro eliminado',
    VALORES_OLD     	TEXT COMMENT 'Concatenación de todos los valores del registro eliminado',
    USUARIO         	VARCHAR(100) NOT NULL COMMENT 'Usuario que realizó la eliminación',
    FECHA_ELIMINACION 	TIMESTAMP DEFAULT CURRENT_TIMESTAMP COMMENT 'Fecha y hora de la eliminación'
) COMMENT='Tabla de auditoría para registros eliminados';

-- TABLAS ESQUEMA VENTAS
-- =======================================================================================================================================================
-- =========================================
-- TABLA: TIPO_DOCUMENTO
-- Catálogo de tipos de documento
-- =========================================
DROP TABLE IF EXISTS TIPO_DOCUMENTO;
CREATE TABLE TIPO_DOCUMENTO (
    ID_TIPO_DOC VARCHAR(10) PRIMARY KEY COMMENT 'Identificador único del tipo de documento',
    NOMBRE_TIPO_DOC VARCHAR(50) NOT NULL COMMENT 'Nombre del tipo de documento (Cédula, RUC, Pasaporte, etc.)',
    DETALLE_TIPO_DOC VARCHAR(100) COMMENT 'Descripción detallada del tipo de documento',
    LONGITUD_DOC INT NOT NULL COMMENT 'Número de dígitos o caracteres esperados',
    FUNCION_VALIDACION VARCHAR(50) NOT NULL COMMENT 'Nombre de la función que valida el documento'
);
-- =========================================
-- TABLA: CLIENTE
-- Registro de clientes con validación de documento y correo
-- =========================================
DROP TABLE IF EXISTS CLIENTE;
CREATE TABLE CLIENTE (
    ID_CLIENTE INT AUTO_INCREMENT PRIMARY KEY COMMENT 'Identificador único del cliente',
    ID_DOCUMENTO VARCHAR(13) NOT NULL UNIQUE COMMENT 'Número de documento único del cliente',
    ID_TIPO_DOC VARCHAR(10) NOT NULL COMMENT 'Tipo de documento, FK hacia TIPO_DOCUMENTO',
    NOMBRES VARCHAR(200) NOT NULL COMMENT 'Nombres completos del cliente',
    EMAIL VARCHAR(100) COMMENT 'Correo electrónico del cliente',
    TELEFONO VARCHAR(100) COMMENT 'Teléfono de contacto',
    DIRECCION VARCHAR(350) COMMENT 'Dirección física del cliente',
    ESTADO CHAR(1) DEFAULT 'A' COMMENT 'Estado del cliente: A=Activo, I=Inactivo, S= Suspendido, P=Pendiente, E= En Espera, O=Obsoleto, R=Reservado, T=Test',
    FECHA_CREACION TIMESTAMP DEFAULT CURRENT_TIMESTAMP COMMENT 'Fecha de creación del registro',
    FECHA_MODIFICACION TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP COMMENT 'Fecha de última modificación',
    CONSTRAINT chk_estado_cliente CHECK (ESTADO IN ('A','I','S','P','E','O','R','T')),
    CONSTRAINT fk_cliente_tipo_doc FOREIGN KEY (ID_TIPO_DOC) REFERENCES TIPO_DOCUMENTO(ID_TIPO_DOC) ON DELETE RESTRICT
);
-- =========================================
-- TABLA: EMPLEADO
-- Registro de empleados asociados a sucursales
-- =========================================
DROP TABLE IF EXISTS EMPLEADO;
CREATE TABLE EMPLEADO (
    ID_EMPLEADO INT AUTO_INCREMENT PRIMARY KEY COMMENT 'Identificador único del empleado',
    ID_DOCUMENTO VARCHAR(13) NOT NULL UNIQUE COMMENT 'Número de documento único del empleado',
    ID_TIPO_DOC VARCHAR(10) NOT NULL COMMENT 'Tipo de documento, FK hacia TIPO_DOCUMENTO (solo Cédula, RUC, Pasaporte)',
    ID_SUCURSAL VARCHAR(10) NOT NULL COMMENT 'Sucursal a la que pertenece el empleado',
    NOMBRES VARCHAR(300) NOT NULL COMMENT 'Nombres completos del empleado',
    TELEFONO VARCHAR(100) COMMENT 'Teléfono de contacto',
    EMAIL VARCHAR(100) COMMENT 'Correo electrónico del empleado',
    DIRECCION VARCHAR(350) COMMENT 'Dirección física del empleado',
    CARGO VARCHAR(100) COMMENT 'Cargo o puesto del empleado',
    FECHA_CONTRATACION DATE NOT NULL COMMENT 'Fecha de contratación',
    ESTADO CHAR(1) DEFAULT 'A' COMMENT 'Estado del empleado: A=Activo, I=Inactivo, S= Suspendido, P=Pendiente, E= En Espera, O=Obsoleto, R=Reservado, T=Test',
    FECHA_CREACION TIMESTAMP DEFAULT CURRENT_TIMESTAMP COMMENT 'Fecha de creación del registro',
    FECHA_MODIFICACION TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP COMMENT 'Fecha de última modificación',
    CONSTRAINT chk_estado_empleado CHECK (ESTADO IN ('A','I','S','P','E','O','R','T')),
    CONSTRAINT fk_empleado_tipo_doc FOREIGN KEY (ID_TIPO_DOC) REFERENCES TIPO_DOCUMENTO(ID_TIPO_DOC) ON DELETE RESTRICT,
    CONSTRAINT fk_empleado_sucursal FOREIGN KEY (ID_SUCURSAL) REFERENCES SUCURSALES(ID_SUCURSAL) ON DELETE RESTRICT
);

-- =========================================
-- TABLA: FORMA_PAGO_LEGAL
-- Catálogo oficial de formas de pago según normativa del SRI
-- =========================================
DROP TABLE IF EXISTS FORMA_PAGO_LEGAL;
CREATE TABLE FORMA_PAGO_LEGAL (
    ID_FPL VARCHAR(10) PRIMARY KEY COMMENT 'Identificador único de la forma de pago legal',
    NOMBRE_FPL VARCHAR(100) NOT NULL COMMENT 'Nombre oficial de la forma de pago según SRI',
    DETALLE_FPL VARCHAR(255) COMMENT 'Descripción detallada de la forma de pago'
);

-- =========================================
-- TABLA: FORMA_PAGO
-- Catálogo interno de formas de pago, vinculado al catálogo oficial del SRI
-- =========================================
DROP TABLE IF EXISTS FORMA_PAGO;

-- ================================================================
-- TRIGGER AUDIT: trg_audit_productos - UPDATE PRODUCTO
-- Auditoria Update productos
-- =================================================================
DROP TRIGGER trg_audit_productos;
DELIMITER $$
CREATE TRIGGER trg_audit_productos
AFTER UPDATE ON PRODUCTOS
FOR EACH ROW
BEGIN
    DECLARE usuario VARCHAR(100);
    SET usuario = CURRENT_USER();
    -- Auditar NOMBRE_PRODUCTO
    IF OLD.NOMBRE_PRODUCTO <> NEW.NOMBRE_PRODUCTO THEN
        INSERT INTO AUDITORIA_UPDATE (NOMBRE_TABLA, CLAVE_PRIMARIA, CAMPO, VALOR_ANTERIOR, VALOR_NUEVO, USUARIO)
        VALUES ('PRODUCTOS', OLD.ID_PRODUCTO, 'NOMBRE_PRODUCTO', OLD.NOMBRE_PRODUCTO, NEW.NOMBRE_PRODUCTO, usuario);
    END IF;
    -- Auditar PVP
    IF OLD.PVP <> NEW.PVP THEN
        INSERT INTO AUDITORIA_UPDATE (NOMBRE_TABLA, CLAVE_PRIMARIA, CAMPO, VALOR_ANTERIOR, VALOR_NUEVO, USUARIO)
        VALUES ('PRODUCTOS', OLD.ID_PRODUCTO, 'PVP', OLD.PVP, NEW.PVP, usuario);
    END IF;
    -- Auditar Estado
    IF OLD.ESTADO <> NEW.ESTADO THEN
        INSERT INTO AUDITORIA_UPDATE (NOMBRE_TABLA, CLAVE_PRIMARIA, CAMPO, VALOR_ANTERIOR, VALOR_NUEVO, USUARIO)
        VALUES ('PRODUCTOS', OLD.ID_PRODUCTO, 'ESTADO', OLD.ESTADO, NEW.ESTADO, usuario);
    END IF;
END$$
DELIMITER ;

-- ================================================================
-- TRIGGER: trg_validar_cliente - INSERT CLIENTE
-- Verificaciones de ID_DOCUMENTO
-- =================================================================
DROP TRIGGER IF EXISTS trg_validar_cliente;
DELIMITER $$
CREATE TRIGGER trg_validar_cliente
BEFORE INSERT ON CLIENTE
FOR EACH ROW
BEGIN
    DECLARE v_funcion VARCHAR(50);
    -- Obtener la función de validación asociada al tipo de documento
    SELECT FUNCION_VALIDACION INTO v_funcion
    FROM TIPO_DOCUMENTO
    WHERE ID_TIPO_DOC = NEW.ID_TIPO_DOC;

    -- Validación de documento según tipo
    IF v_funcion = 'fn_validar_cedula' AND fn_validar_cedula(NEW.ID_DOCUMENTO) = 0 THEN
        SIGNAL SQLSTATE '45000' SET MESSAGE_TEXT = 'Cédula inválida';
    END IF;

    IF v_funcion = 'fn_validar_ruc' AND fn_validar_ruc(NEW.ID_DOCUMENTO) = 0 THEN
        SIGNAL SQLSTATE '45000' SET MESSAGE_TEXT = 'RUC inválido';
    END IF;

    IF v_funcion = 'fn_validar_pasaporte' AND fn_validar_pasaporte(NEW.ID_DOCUMENTO) = 0 THEN
        SIGNAL SQLSTATE '45000' SET MESSAGE_TEXT = 'Pasaporte inválido';
    END IF;

    /*IF v_funcion = 'fn_validar_placa' AND fn_validar_placa(NEW.ID_DOCUMENTO) = 0 THEN
        SIGNAL SQLSTATE '45000' SET MESSAGE_TEXT = 'Placa inválida';
    END IF;*/ -- Definido pero No Aplica al momento

    IF v_funcion = 'fn_validar_consumidor_final' AND fn_validar_consumidor_final(NEW.ID_DOCUMENTO) = 0 THEN
        SIGNAL SQLSTATE '45000' SET MESSAGE_TEXT = 'Documento consumidor final inválido';
    END IF;

    -- Validación de correo electrónico (si no es nulo)
    IF NEW.EMAIL IS NOT NULL AND fn_validar_email(NEW.EMAIL) = 0 THEN
        SIGNAL SQLSTATE '45000' SET MESSAGE_TEXT = 'Correo electrónico inválido';
    END IF;
END$$
DELIMITER ;

DELIMITER $$
CREATE TRIGGER trg_validar_empleado
BEFORE INSERT ON EMPLEADO
FOR EACH ROW
BEGIN
    DECLARE v_funcion VARCHAR(50);

    -- Validar que el tipo de documento sea solo CÉDULA, RUC o PASAPORTE
    IF NEW.ID_TIPO_DOC NOT IN ('TIP_DOC01','TIP_DOC02','TIP_DOC03') THEN
        SIGNAL SQLSTATE '45000' SET MESSAGE_TEXT = 'Tipo de documento inválido para empleado';
    END IF;

    -- Obtener la función de validación asociada al tipo de documento
    SELECT FUNCION_VALIDACION INTO v_funcion
    FROM TIPO_DOCUMENTO
    WHERE ID_TIPO_DOC = NEW.ID_TIPO_DOC;

    -- Validación de documento según tipo
    IF v_funcion = 'fn_validar_cedula' AND fn_validar_cedula(NEW.ID_DOCUMENTO) = 0 THEN
        SIGNAL SQLSTATE '45000' SET MESSAGE_TEXT = 'Cédula inválida';
    END IF;

    IF v_funcion = 'fn_validar_ruc' AND fn_validar_ruc(NEW.ID_DOCUMENTO) = 0 THEN
        SIGNAL SQLSTATE '45000' SET MESSAGE_TEXT = 'RUC inválido';
    END IF;

    IF v_funcion = 'fn_validar_pasaporte' AND fn_validar_pasaporte(NEW.ID_DOCUMENTO) = 0 THEN
        SIGNAL SQLSTATE '45000' SET MESSAGE_TEXT = 'Pasaporte inválido';
    END IF;

    -- Validación de correo electrónico (si no es nulo)
    IF NEW.EMAIL IS NOT NULL AND fn_validar_email(NEW.EMAIL) = 0 THEN
        SIGNAL SQLSTATE '45000' SET MESSAGE_TEXT = 'Correo electrónico inválido';
    END IF;
END$$

-- =========================================================

DELIMITER $$

BEFORE INSERT ON TRANSFERENCIA
FOR EACH ROW
BEGIN
    DECLARE v_stock_actual INT;
    -- Obtener stock actual del producto en la sucursal origen
    SELECT CANTIDAD INTO v_stock_actual
    FROM PRODUCTOS_SUCURSAL
    WHERE ID_PRODUCTO = NEW.ID_PRODUCTO
      AND ID_SUCURSAL = NEW.SUC_ORIGEN;

    -- Validar existencia de stock
    IF v_stock_actual IS NULL THEN
        SIGNAL SQLSTATE '45000'
        SET MESSAGE_TEXT = 'El producto no existe en la sucursal origen';
    END IF;

    -- Validar cantidad suficiente
    IF v_stock_actual < NEW.CANTIDAD THEN
        SIGNAL SQLSTATE '45000'
        SET MESSAGE_TEXT = 'Stock insuficiente en la sucursal origen';
    END IF;

    -- Si pasa la validación, se descuenta stock en origen y se suma en destino
    UPDATE PRODUCTOS_SUCURSAL
    SET CANTIDAD = CANTIDAD - NEW.CANTIDAD
    WHERE ID_PRODUCTO = NEW.ID_PRODUCTO
      AND ID_SUCURSAL = NEW.SUC_ORIGEN;

    UPDATE PRODUCTOS_SUCURSAL
    SET CANTIDAD = CANTIDAD + NEW.CANTIDAD
    WHERE ID_PRODUCTO = NEW.ID_PRODUCTO
      AND ID_SUCURSAL = NEW.SUC_DESTINO;
END$$

DELIMITER $$
CREATE FUNCTION fn_validar_pasaporte(p_pasaporte VARCHAR(9))
RETURNS TINYINT
DETERMINISTIC
BEGIN
    DECLARE longitudPasaporte INT;

    -- Normalizar: eliminar espacios y convertir a mayúsculas
    SET p_pasaporte = UPPER(TRIM(p_pasaporte));
    SET longitudPasaporte = LENGTH(p_pasaporte);

    -- Validar longitud exacta
    IF longitudPasaporte <> 9 THEN
        RETURN 0;
    END IF;

    -- Validar formato: 3 letras + 6 números
    IF p_pasaporte REGEXP '^[A-Z]{3}[0-9]{6}$' THEN
        RETURN 1; -- Válido
    ELSE
        RETURN 0; -- Inválido
    END IF;
END$$

-- ==============================================
-- FUNCION: fn_validar_ruc
-- Verificacion RUC valido Modulo 10 y Modulo 11
-- ==============================================
DROP FUNCTION IF EXISTS fn_validar_ruc;
DELIMITER $$
CREATE FUNCTION fn_validar_ruc(p_ruc VARCHAR(13))
RETURNS TINYINT
DETERMINISTIC
BEGIN
    DECLARE provincia INT;
    DECLARE tercerDigito INT;
    DECLARE establecimiento VARCHAR(4);
    DECLARE suma INT DEFAULT 0;
    DECLARE digito INT;
    DECLARE verificadorReal INT;
    DECLARE verificadorCalculado INT;
    DECLARE residuo INT;
    DECLARE i INT;

    -- Validar longitud exacta de 13 dígitos
    IF p_ruc REGEXP '^[0-9]{13}$' = 0 THEN
        RETURN 0;
    END IF;

    -- Provincia
    SET provincia = CAST(SUBSTRING(p_ruc,1,2) AS UNSIGNED);
    IF (provincia < 1 OR provincia > 24) AND provincia <> 30 THEN
        RETURN 0;
    END IF;

    -- Tercer dígito
    SET tercerDigito = CAST(SUBSTRING(p_ruc,3,1) AS UNSIGNED);

    -- CASO A: Persona Natural (0–5)
    IF tercerDigito BETWEEN 0 AND 5 THEN
        SET establecimiento = SUBSTRING(p_ruc,11,3);
        IF establecimiento = '000' THEN
            RETURN 0;
        END IF;

        SET suma = 0;
        SET i = 1;
        WHILE i <= 9 DO
            SET digito = CAST(SUBSTRING(p_ruc,i,1) AS UNSIGNED);
            IF MOD(i,2)=1 THEN
                SET digito = digito * 2;
                IF digito >= 10 THEN SET digito = digito - 9; END IF;
            END IF;
            SET suma = suma + digito;
            SET i = i + 1;
        END WHILE;

        SET verificadorReal = CAST(SUBSTRING(p_ruc,10,1) AS UNSIGNED);
        SET verificadorCalculado = (10 - (suma MOD 10)) MOD 10;

        RETURN IF(verificadorCalculado = verificadorReal, 1, 0);
    END IF;

    -- CASO B: Sociedad Privada / Extranjeros (9)
    IF tercerDigito = 9 THEN
        SET establecimiento = SUBSTRING(p_ruc,11,3);
        IF establecimiento = '000' THEN
            RETURN 0;
        END IF;

        SET suma = 0;
        SET i = 1;
        WHILE i <= 9 DO
            SET digito = CAST(SUBSTRING(p_ruc,i,1) AS UNSIGNED);
            CASE i
                WHEN 1 THEN SET suma = suma + digito*4;
                WHEN 2 THEN SET suma = suma + digito*3;
                WHEN 3 THEN SET suma = suma + digito*2;
                WHEN 4 THEN SET suma = suma + digito*7;
                WHEN 5 THEN SET suma = suma + digito*6;
                WHEN 6 THEN SET suma = suma + digito*5;
                WHEN 7 THEN SET suma = suma + digito*4;
                WHEN 8 THEN SET suma = suma + digito*3;
                WHEN 9 THEN SET suma = suma + digito*2;
            END CASE;
            SET i = i + 1;
        END WHILE;

        SET verificadorReal = CAST(SUBSTRING(p_ruc,10,1) AS UNSIGNED);
        SET residuo = suma MOD 11;
        SET verificadorCalculado = IF(residuo=0,0,11-residuo);

        RETURN IF(verificadorCalculado = verificadorReal, 1, 0);
    END IF;

    -- CASO C: Instituciones Públicas (6)
    IF tercerDigito = 6 THEN
        SET establecimiento = SUBSTRING(p_ruc,10,4);
        IF establecimiento = '0000' THEN
            RETURN 0;
        END IF;

        SET suma = 0;
        SET i = 1;
        WHILE i <= 8 DO
            SET digito = CAST(SUBSTRING(p_ruc,i,1) AS UNSIGNED);
            CASE i
                WHEN 1 THEN SET suma = suma + digito*3;
                WHEN 2 THEN SET suma = suma + digito*2;
                WHEN 3 THEN SET suma = suma + digito*7;
                WHEN 4 THEN SET suma = suma + digito*6;
                WHEN 5 THEN SET suma = suma + digito*5;
                WHEN 6 THEN SET suma = suma + digito*4;
                WHEN 7 THEN SET suma = suma + digito*3;
                WHEN 8 THEN SET suma = suma + digito*2;
            END CASE;
            SET i = i + 1;
        END WHILE;

        SET verificadorReal = CAST(SUBSTRING(p_ruc,9,1) AS UNSIGNED);
        SET residuo = suma MOD 11;
        SET verificadorCalculado = IF(residuo=0,0,11-residuo);

        RETURN IF(verificadorCalculado = verificadorReal, 1, 0);
    END IF;

    -- Si no cumple ningún caso válido
    RETURN 0;
END$$
DELIMITER ;

-- =========================================
-- FUNCION: fn_validar_cedula
-- Algoritmo modulo 10 para validar cedula
-- =========================================
DROP FUNCTION IF EXISTS fn_validar_cedula;
DELIMITER $$
CREATE FUNCTION fn_validar_cedula(p_cedula VARCHAR(10))
RETURNS TINYINT
DETERMINISTIC
BEGIN
    DECLARE i INT;
    DECLARE suma INT DEFAULT 0;
    DECLARE digito INT;
    DECLARE coef INT;
    DECLARE prod INT;
    DECLARE verificadorCalculado INT;
    DECLARE verificadorReal INT;
    DECLARE provincia INT;
    DECLARE tercerDigito INT;

    -- Validar longitud
    IF LENGTH(p_cedula) <> 10 THEN
        RETURN 0;
    END IF;

    -- Provincia (primeros 2 dígitos)
    SET provincia = CAST(SUBSTRING(p_cedula, 1, 2) AS UNSIGNED);
    IF (provincia < 1 OR provincia > 24) AND provincia <> 30 THEN
        RETURN 0;
    END IF;

    -- Tercer dígito
    SET tercerDigito = CAST(SUBSTRING(p_cedula, 3, 1) AS UNSIGNED);
    IF tercerDigito >= 6 THEN
        RETURN 0;
    END IF;

    -- Calcular suma con coeficientes
    SET i = 1;
    WHILE i <= 9 DO
        SET digito = CAST(SUBSTRING(p_cedula, i, 1) AS UNSIGNED);

        IF MOD(i,2) = 1 THEN
            SET coef = 2;
        ELSE
            SET coef = 1;
        END IF;

        SET prod = digito * coef;
        IF prod >= 10 THEN
            SET prod = prod - 9;
        END IF;

        SET suma = suma + prod;
        SET i = i + 1;
    END WHILE;

    -- Dígito verificador
    SET verificadorReal = CAST(SUBSTRING(p_cedula, 10, 1) AS UNSIGNED);
    SET verificadorCalculado = (10 - (suma MOD 10)) MOD 10;

    IF verificadorCalculado = verificadorReal THEN
        RETURN 1; -- Válida
    ELSE
        RETURN 0; -- Inválida
    END IF;
END$$
DELIMITER ;

DROP PROCEDURE IF EXISTS next_document_number;
DELIMITER $$
CREATE PROCEDURE next_document_number(IN doc_type VARCHAR(20), IN series VARCHAR(20), OUT next_num VARCHAR(50))
BEGIN
  DECLARE curr INT;
  -- Obtener el número actual
  SELECT current_number INTO curr
  FROM document_numbers
  WHERE doc_type = doc_type AND series = series
  LIMIT 1;
  -- Si no existe, inicializar
  IF curr IS NULL THEN
    SET curr = 0;
    INSERT INTO document_numbers (doc_type, series, current_number)
    VALUES (doc_type, series, curr);
  END IF;
  -- Incrementar
  SET curr = curr + 1;
  UPDATE document_numbers
  SET current_number = curr
  WHERE doc_type = doc_type AND series = series;
  -- Retornar número formateado
  SET next_num = CONCAT(series, '-', LPAD(curr, 8, '0'));
END$$
DELIMITER ;

  P.DESCRIPCION       	AS descripcion,
  P.ID_SUBCATEGORIA		AS idSubcategoria,
  C.NOMBRE_CATEGORIA  	AS categoria,
  SC.NOMBRE           	AS subcategoria,
  P.ETIQUETAS         	AS etiquetas,
  P.UNIDAD_MEDIDA     	AS unidadMedida,
  P.PESO_VOLUMEN      	AS pesoVolumen,
  SM.ID_STOCK			AS idStockMinimo,
  SM.CANTIDAD          	AS stockMin,
  SX.ID_STOCK			AS idStockMax,
  SX.CANTIDAD			AS stockMaximo,
  PS.CANTIDAD         	AS cantidad
FROM PRODUCTOS p
LEFT JOIN STOCK SM ON P.ID_STOCK_MIN=SM.ID_STOCK AND SM.TIPO_STOCK='M' AND SM.ESTADO='A'
LEFT JOIN STOCK SX ON P.ID_STOCK_MAX=SX.ID_STOCK AND SX.TIPO_STOCK='X' AND SX.ESTADO='A'
JOIN PRODUCTOS_SUCURSAL PS ON P.ID_PRODUCTO = PS.ID_PRODUCTO 
JOIN SUCURSALES SR ON PS.ID_SUCURSAL=SR.ID_SUCURSAL
JOIN SUBCATEGORIAS SC ON p.ID_SUBCATEGORIA=SC.ID_SUBCATEGORIA
JOIN CATEGORIAS C ON SC.ID_CATEGORIA=C.ID_CATEGORIA
WHERE P.ESTADO='A'
ORDER BY P.ID_PRODUCTO,SR.ID_SUCURSAL;

-- =========================================
-- VISTA: v_ventas_detalle
-- Todas las ventas con su detalle de productos
-- =========================================
DROP VIEW IF EXISTS v_ventas_detalle;
CREATE VIEW v_ventas_detalle AS
SELECT 
  v.ID_VENTA          	AS idVenta,
  v.FECHA_HORA        	AS fechaHora,
  v.SUBTOTAL          	AS subtotal,
  V.DESCUENTO			AS descuento,
  CONCAT(TRUNCATE(I.VALOR_IMPUESTO,0),'%')	AS impuesto,
  V.VALOR_IMPUESTO    	AS valorImpuesto,
  V.TOTAL				AS total,
  C.NOMBRES 			AS cliente,
  E.NOMBRES				AS empleado,
  S.ID_SUCURSAL			AS idSucursal,
  S.NOMBRE_SUCURSAL		AS sucursal,
  FP.NOMBRE_FORMA_PAGO	AS formaPago,
  G.REFERENCIA			AS referenciaPago,
  G.MONTO				AS montoPago,
  G.ESTADO				AS estadoPago,
  P.ID_PRODUCTO			AS idProducto,
  P.NOMBRE_PRODUCTO		AS nombreProducto,
  P.PVP					AS precioProducto,
  CC.NOMBRE_CATEGORIA	AS nombreCategoria,
  DV.CANTIDAD			AS cantidad
FROM VENTA v
JOIN IMPUESTO I ON V.ID_TIPO_IMPUESTO=I.ID_TIPO_IMPUESTO
JOIN CLIENTE C ON V.ID_CLIENTE=C.ID_CLIENTE
JOIN EMPLEADO E ON V.ID_EMPLEADO=E.ID_EMPLEADO
JOIN SUCURSALES S ON V.ID_SUCURSAL=S.ID_SUCURSAL
JOIN PAGO G ON V.ID_PAGO=G.ID_PAGO
JOIN FORMA_PAGO FP ON G.ID_FORMA_PAGO=FP.ID_FORMA_PAGO
JOIN DETALLE_VENTA DV ON V.ID_VENTA=DV.ID_VENTA
JOIN PRODUCTOS P ON P.ID_PRODUCTO=DV.ID_PRODUCTO
JOIN SUBCATEGORIAS SC ON P.ID_SUBCATEGORIA=SC.ID_SUBCATEGORIA
JOIN CATEGORIAS CC ON CC.ID_CATEGORIA=SC.ID_CATEGORIA
WHERE V.ESTADO = 'A'
ORDER BY V.ID_VENTA,P.ID_PRODUCTO;

-- Ventas unicas sin detalle de productos
-- =========================================
DROP VIEW IF EXISTS v_ventas;
CREATE VIEW v_ventas AS
SELECT 
  v.ID_VENTA          	AS idVenta,
  v.FECHA_HORA        	AS fechaHora,
  v.SUBTOTAL          	AS subtotal,
  V.DESCUENTO			AS descuento,
  CONCAT(TRUNCATE(I.VALOR_IMPUESTO,0),'%')	AS impuesto,
  V.VALOR_IMPUESTO    	AS valorImpuesto,
  V.TOTAL				AS total,
  C.NOMBRES 			AS cliente,
  E.NOMBRES				AS empleado,
  S.ID_SUCURSAL			AS idSucursal,
  S.NOMBRE_SUCURSAL		AS sucursal,
  FP.NOMBRE_FORMA_PAGO	AS formaPago,
  G.REFERENCIA			AS referenciaPago,
  G.MONTO				AS montoPago,
  G.ESTADO				AS estadoPago
FROM VENTA v
JOIN IMPUESTO I ON V.ID_TIPO_IMPUESTO=I.ID_TIPO_IMPUESTO
JOIN CLIENTE C ON V.ID_CLIENTE=C.ID_CLIENTE
JOIN EMPLEADO E ON V.ID_EMPLEADO=E.ID_EMPLEADO
JOIN SUCURSALES S ON V.ID_SUCURSAL=S.ID_SUCURSAL

WHERE V.ESTADO = 'A'
